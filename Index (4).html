require(lot.seller == msg.sender || msg.sender == auctioneer, "ApexAuction: Unauthorized"); require(lot.state == LotState.Active, "ApexAuction: Not active"); lot.state = LotState.Canceled; uint256 refund = lot.highestBid; require(sent, "ApexAuction: Refund failed"); } emit AuctionCanceled(lotId); } // 2. createAuctionLot function createAuctionLot( uint256 reservePrice, uint256 durationSeconds, bytes32 itemHash ) external returns (uint256) { require(durationSeconds >= 60, "ApexAuction: Min duration 60s"); require(lotId > 0 && lotId <= lotCounter, "ApexAuction: Invalid lot ID"); require(lot.state == LotState.Active, "ApexAuction: Not active"); lot.state = LotState.Settled; uint256 totalPayout = lot.highestBid; if (totalPayout > 0) { uint256 fee = (totalPayout * platformFeeBps) / 10000; uint256 netSellerAmount = totalPayout - fee; accumulatedFees += fee; (bool sent, ) = lot.seller.call{value: netSellerAmount}(""); } // 5. placeBid function placeBid(uint256 lotId) external payable { AuctionLot storage lot = lots[lotId]; require(lotId > 0 && lotId <= lotCounter, "ApexAuction: Invalid ID"); require(lot.state == LotState.Active, "ApexAuction: Not active"); } lot.highestBid = msg.value; lot.highestBidder = payable(msg.sender); emit BidPlaced(lotId, msg.sender, msg.value); } // 6. reclaimOverbid function reclaimOverbid(uint256 lotId) external { uint256 amount = pendingReturns[lotId][msg.sender]; require(amount > 0, "ApexAuction: No funds to reclaim"); pendingReturns[lotId][msg.sender] = 0; (bool sent, ) = payable(msg.sender).call{value: amount}(""); require(sent, "ApexAuction: Reclaim failed"); emit OverbidRefunded(lotId, msg.sender, amount); } // 7. updateFeeCut function updateFeeCut(uint256 newFeeBps) external onlyAuctioneer { require(newFeeBps <= 1000, "ApexAuction: Max fee 10%"); platformFeeBps = newFeeBps;
